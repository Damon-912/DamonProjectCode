# 系统管理模块设计文档

## 1. 功能概述

系统管理模块提供用户管理、角色权限、菜单配置、接口服务配置等功能，支持系统的日常运维和权限控制。

## 3. 后端接口设计

### 3.1 查询用户列表（Code: 02010401）

```objectscript
ClassMethod QuerySystemUsers(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"","result":{"total":0,"rows":[]}}
    
    try {
        set params = jsonObj.params.%Get(0)
        set userName = params.userName
        set deptCode = params.deptCode
        set status = params.status
        
        set pageSize = 50, currentPage = 1
        if jsonObj.%IsDefined("pagination") {
            set pagObj = jsonObj.pagination.%Get(0)
            set pageSize = pagObj.pageSize
            set currentPage = pagObj.currentPage
        }
        
        set sql = "SELECT ID, UserId, UserName, LoginCode, DeptCode, "
                 _"Phone, Email, Status, CreateTime, LastLoginTime "
                 _"FROM BS_DRGSystemUser WHERE 1=1"
        
        if userName '= "" {
            set sql = sql _ " AND UserName LIKE '%"_userName_"%'"
        }
        if deptCode '= "" {
            set sql = sql _ " AND DeptCode = '"_deptCode_"'"
        }
        if status '= "" {
            set sql = sql _ " AND Status = '"_status_"'"
        }
        set sql = sql _ " ORDER BY CreateTime DESC"
        
        set stmt = ##class(%SQL.Statement).%New()
        do stmt.%Prepare(sql)
        set rs = stmt.%Execute()
        
        set count = 0, dataArr = []
        while rs.%Next() {
            set count = count + 1
            continue:(count < ((currentPage-1)*pageSize+1)) || (count > (currentPage*pageSize))
            
            set dataObj = {}
            set dataObj.userId = rs.%Get("UserId")
            set dataObj.userName = rs.%Get("UserName")
            set dataObj.loginCode = rs.%Get("LoginCode")
            set dataObj.deptCode = rs.%Get("DeptCode")
            set dataObj.phone = rs.%Get("Phone")
            set dataObj.email = rs.%Get("Email")
            set dataObj.status = rs.%Get("Status")
            set dataObj.createTime = rs.%Get("CreateTime")
            set dataObj.lastLoginTime = rs.%Get("LastLoginTime")
            do dataArr.%Push(dataObj)
        }
        
        set resultObj.result.total = count
        set resultObj.result.rows = dataArr
    }
    catch(e) {
        set resultObj.errorCode = "02010401"
        set resultObj.errorMessage = "查询用户失败"
    }
    
    quit resultObj
}
```

### 3.2 保存用户（Code: 02010402）

```objectscript
ClassMethod SaveSystemUser(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"保存成功"}
    
    try {
        set params = jsonObj.params.%Get(0)
        set userId = params.userId
        
        set obj = ##class(User.BSDRGSystemUser).%OpenId(userId)
        if obj = "" {
            set obj = ##class(User.BSDRGSystemUser).%New()
            set obj.UserId = userId
        }
        
        set obj.UserName = params.userName
        set obj.LoginCode = params.loginCode
        if params.password '= "" {
            set obj.Password = $System.Encryption.SHA1Hash(params.password)
        }
        set obj.DeptCode = params.deptCode
        set obj.Phone = params.phone
        set obj.Email = params.email
        set obj.Status = params.status
        
        set sc = obj.%Save()
        if $$$ISERR(sc) {
            set resultObj.errorCode = "02010402"
            set resultObj.errorMessage = "保存失败"
        }
    }
    catch(e) {
        set resultObj.errorCode = "02010402"
        set resultObj.errorMessage = "保存用户异常"
    }
    
    quit resultObj
}
```

## 4. 前端页面设计

### 4.1 接口服务配置页面

```typescript
// pages/System/Interface/index.tsx
import React, { useState } from 'react';
import { Card, Table, Button, Input, Select, Tag, Modal, Form, message } from 'antd';
import { queryInterfaceServices, saveInterfaceService } from '@/api/system';
import { useTable } from '@/hooks/useTable';

const { Option } = Select;

const InterfaceConfig: React.FC = () => {
  const [isModalVisible, setIsModalVisible] = useState(false);
  const [form] = Form.useForm();
  const [editingRecord, setEditingRecord] = useState<any>(null);
  
  const { tableProps, refresh } = useTable({
    api: queryInterfaceServices,
    defaultParams: { pageSize: 50, currentPage: 1 }
  });

  const columns = [
    { title: 'Code', dataIndex: 'code', width: 100 },
    { title: '描述', dataIndex: 'descripts' },
    { title: '类名', dataIndex: 'className', width: 200 },
    { title: '方法名', dataIndex: 'methodName', width: 150 },
    { 
      title: '类型', 
      dataIndex: 'serviceType',
      render: (type) => {
        const typeMap = { S: '查询', A: '新增', U: '修改', D: '删除' };
        return <Tag>{typeMap[type] || type}</Tag>;
      }
    },
    { title: '生效日期', dataIndex: 'startDate' },
    { title: '失效日期', dataIndex: 'stopDate' },
    {
      title: '操作',
      render: (_, record) => (
        <Button type="link" onClick={() => handleEdit(record)}>编辑</Button>
      )
    }
  ];

  const handleEdit = (record: any) => {
    setEditingRecord(record);
    form.setFieldsValue(record);
    setIsModalVisible(true);
  };

  const handleSave = async () => {
    const values = await form.validateFields();
    const result = await saveInterfaceService({
      id: editingRecord?.id,
      ...values
    });
    if (result.errorCode === '0') {
      message.success('保存成功');
      setIsModalVisible(false);
      refresh();
    } else {
      message.error(result.errorMessage);
    }
  };

  return (
    <Card title="接口服务配置" extra={
      <Button type="primary" onClick={() => { setEditingRecord(null); form.resetFields(); setIsModalVisible(true); }}>
        新增接口
      </Button>
    }>
      <Table {...tableProps} columns={columns} rowKey="id" />
      
      <Modal
        title={editingRecord ? '编辑接口' : '新增接口'}
        visible={isModalVisible}
        onOk={handleSave}
        onCancel={() => setIsModalVisible(false)}
        width={600}
      >
        <Form form={form} layout="vertical">
          <Form.Item name="code" label="Code" rules={[{ required: true }]}>
            <Input placeholder="如：02010101" disabled={!!editingRecord} />
          </Form.Item>
          <Form.Item name="descripts" label="描述" rules={[{ required: true }]}>
            <Input />
          </Form.Item>
          <Form.Item name="className" label="类名" rules={[{ required: true }]}>
            <Input placeholder="如：src.DRG.Warning" />
          </Form.Item>
          <Form.Item name="methodName" label="方法名" rules={[{ required: true }]}>
            <Input />
          </Form.Item>
          <Form.Item name="serviceType" label="服务类型" rules={[{ required: true }]}>
            <Select>
              <Option value="S">查询</Option>
              <Option value="A">新增</Option>
              <Option value="U">修改</Option>
              <Option value="D">删除</Option>
            </Select>
          </Form.Item>
        </Form>
      </Modal>
    </Card>
  );
};

export default InterfaceConfig;
```
