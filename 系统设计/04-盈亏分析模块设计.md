# 盈亏分析模块设计文档

## 1. 功能概述

盈亏分析模块计算和分析DRG/DIP付费模式下病案、科室、医生、病种的盈亏情况，提供多维度统计报表。

## 2. 数据表设计

### 2.1 盈亏分析表（BS_DRGProfitAnalysis）

```objectscript
Class User.BSDRGProfitAnalysis Extends (%Persistent, %XML.Adaptor)
```

## 3. 后端接口设计

### 3.1 查询科室盈亏（Code: 02010201）

```objectscript
ClassMethod QueryDeptProfit(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"","result":[]}
    
    try {
        set params = jsonObj.params.%Get(0)
        set year = params.year
        set month = params.month
        set deptCode = params.deptCode
        
        set sql = "SELECT DeptCode, DeptName, Year, Month, TotalCases, "
                 _"TotalPayment, TotalActualCost, TotalProfit, ProfitRate, "
                 _"ProfitCases, LossCases "
                 _"FROM BS_DRGDeptSummary WHERE Year = ? AND Month = ?"
        
        if deptCode '= "" {
            set sql = sql _ " AND DeptCode = ?"
        }
        set sql = sql _ " ORDER BY TotalProfit DESC"
        
        set stmt = ##class(%SQL.Statement).%New()
        do stmt.%Prepare(sql)
        set rs = stmt.%Execute(year, month, deptCode)
        
        set dataArr = []
        while rs.%Next() {
            set dataObj = {}
            set dataObj.deptCode = rs.%Get("DeptCode")
            set dataObj.deptName = rs.%Get("DeptName")
            set dataObj.year = rs.%Get("Year")
            set dataObj.month = rs.%Get("Month")
            set dataObj.totalCases = rs.%Get("TotalCases")
            set dataObj.totalPayment = rs.%Get("TotalPayment")
            set dataObj.totalActualCost = rs.%Get("TotalActualCost")
            set dataObj.totalProfit = rs.%Get("TotalProfit")
            set dataObj.profitRate = rs.%Get("ProfitRate")
            set dataObj.profitCases = rs.%Get("ProfitCases")
            set dataObj.lossCases = rs.%Get("LossCases")
            do dataArr.%Push(dataObj)
        }
        
        set resultObj.result = dataArr
    }
    catch(e) {
        set resultObj.errorCode = "02010201"
        set resultObj.errorMessage = "查询科室盈亏失败"
    }
    
    quit resultObj
}
```

### 3.2 查询医生盈亏（Code: 02010202）

```objectscript
ClassMethod QueryDoctorProfit(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"","result":[]}
    
    try {
        set params = jsonObj.params.%Get(0)
        set startDate = params.startDate
        set endDate = params.endDate
        set deptCode = params.deptCode
        
        set sql = "SELECT DoctorName, DeptName, COUNT(*) as CaseCount, "
                 _"SUM(PaymentStandard) as TotalPayment, "
                 _"SUM(ActualCost) as TotalCost, "
                 _"SUM(ProfitAmount) as TotalProfit, "
                 _"AVG(ProfitRate) as AvgProfitRate "
                 _"FROM BS_DRGProfitAnalysis "
                 _"WHERE DscgDate >= ? AND DscgDate <= ?"
        
        if deptCode '= "" {
            set sql = sql _ " AND DeptCode = ?"
        }
        set sql = sql _ " GROUP BY DoctorName, DeptName ORDER BY TotalProfit DESC"
        
        set stmt = ##class(%SQL.Statement).%New()
        do stmt.%Prepare(sql)
        set rs = stmt.%Execute(startDate, endDate, deptCode)
        
        set dataArr = []
        while rs.%Next() {
            set dataObj = {}
            set dataObj.doctorName = rs.%Get("DoctorName")
            set dataObj.deptName = rs.%Get("DeptName")
            set dataObj.caseCount = rs.%Get("CaseCount")
            set dataObj.totalPayment = rs.%Get("TotalPayment")
            set dataObj.totalCost = rs.%Get("TotalCost")
            set dataObj.totalProfit = rs.%Get("TotalProfit")
            set dataObj.avgProfitRate = rs.%Get("AvgProfitRate")
            do dataArr.%Push(dataObj)
        }
        
        set resultObj.result = dataArr
    }
    catch(e) {
        set resultObj.errorCode = "02010202"
        set resultObj.errorMessage = "查询医生盈亏失败"
    }
    
    quit resultObj
```

## 4. 前端页面设计

```typescript
// pages/Profit/Dept/index.tsx
import React from 'react';
import { Card, Table, DatePicker, Select, Statistic, Row, Col } from 'antd';
import { queryDeptProfit } from '@/api/profit';
import { useTable } from '@/hooks/useTable';
import ReactECharts from 'echarts-for-react';

const { Option } = Select;

const DeptProfit: React.FC = () => {
  const { tableProps, data } = useTable({
    api: queryDeptProfit,
    defaultParams: { year: 2024, month: 1 }
  });

  const columns = [
    { title: '排名', render: (_, __, index) => index + 1, width: 60 },
    { title: '科室', dataIndex: 'deptName' },
    { title: '病例数', dataIndex: 'totalCases' },
    { title: '付费标准', dataIndex: 'totalPayment', render: v => `¥${v?.toFixed(2)}` },
    { title: '实际成本', dataIndex: 'totalActualCost', render: v => `¥${v?.toFixed(2)}` },
    { 
      title: '盈亏金额', 
      dataIndex: 'totalProfit',
      render: v => <span style={{ color: v >= 0 ? '#52c41a' : '#f5222d' }}>
        {v >= 0 ? '+' : ''}¥{v?.toFixed(2)}
      </span>
    },
    { 
      title: '盈亏率', 
      dataIndex: 'profitRate',
      render: v => <span style={{ color: v >= 0 ? '#52c41a' : '#f5222d' }}>
        {v >= 0 ? '+' : ''}{v?.toFixed(2)}%
      </span>
    },
    { title: '结余病例', dataIndex: 'profitCases' },
    { title: '亏损病例', dataIndex: 'lossCases' }
  ];

  const chartOption = {
    tooltip: { trigger: 'axis' },
    xAxis: { type: 'category', data: data?.map(d => d.deptName) },
    yAxis: { type: 'value', name: '盈亏金额(万元)' },
    series: [{
      data: data?.map(d => (d.totalProfit / 10000).toFixed(2)),
      type: 'bar',
      itemStyle: {
        color: (params: any) => params.value >= 0 ? '#52c41a' : '#f5222d'
      }
    }]
  };

  return (
    <>
      {/* 统计卡片 */}
      <Row gutter={16} style={{ marginBottom: 16 }}>
        <Col span={6}>
          <Card><Statistic title="总病例数" value={data?.reduce((a, b) => a + b.totalCases, 0)} /></Card>
        </Col>
        <Col span={6}>
          <Card><Statistic 
            title="总盈亏" 
            value={data?.reduce((a, b) => a + b.totalProfit, 0) / 10000} 
            suffix="万元"
            valueStyle={{ color: '#52c41a' }}
          /></Card>
        </Col>
        <Col span={6}>
          <Card><Statistic title="结余科室" value={data?.filter(d => d.totalProfit > 0).length} /></Card>
        </Col>
        <Col span={6}>
          <Card><Statistic title="亏损科室" value={data?.filter(d => d.totalProfit < 0).length} /></Card>
        </Col>
      </Row>

      {/* 图表 */}
      <Card title="科室盈亏分布" style={{ marginBottom: 16 }}>
        <ReactECharts option={chartOption} style={{ height: 300 }} />
      </Card>

      {/* 表格 */}
      <Card title="科室盈亏明细">
        <Table {...tableProps} columns={columns} rowKey="deptCode" />
      </Card>
    </>
  );
};

export default DeptProfit;
```
