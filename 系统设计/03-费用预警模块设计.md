# 费用预警模块设计文档

## 1. 功能概述

费用预警模块实时监控病案费用与DRG/DIP付费标准的偏差，当费用超出预警阈值时触发预警通知。

## 2. 数据表设计

### 2.1 预警规则表（BS_DRGWarningRule）

```objectscript
Class User.BSDRGWarningRule Extends (%Persistent, %XML.Adaptor)
{
    /// 规则编码
    Property RuleCode As %String(MAXLEN = 20) [ Required, Unique ];
    
    /// 规则名称
    Property RuleName As %String(MAXLEN = 50) [ Required ];
    
    /// 规则类型（超支预警/结余预警/费用异常）
    Property RuleType As %String(MAXLEN = 20) [ Required ];
    
    /// 适用DRG（空表示全部）
    Property DrgCode As %String(MAXLEN = 10);
    
    /// 适用科室（空表示全部）
    Property DeptCode As %String(MAXLEN = 20);
    
    /// 阈值百分比
    Property ThresholdPercent As %Float [ Required ];
    
    /// 预警级别（严重/警告/提示）
    Property WarningLevel As %String(MAXLEN = 10) [ Required ];
    
    /// 状态（启用/禁用）
    Property Status As %String(MAXLEN = 10) [ Required ];
    
    /// 创建时间
    Property CreateTime As %TimeStamp [ InitialExpression = {$ZDateTime($H, 3)} ];
    
    /// 更新时间

}
```

## 3. 后端接口设计

### 3.1 查询预警规则（Code: 02010101）

```objectscript
ClassMethod QueryWarningRules(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"","result":{"total":0,"rows":[]}}
    
    try {
        set params = jsonObj.params.%Get(0)
        set ruleType = params.ruleType
        set ruleName = params.ruleName
        set status = params.status
        
        set pageSize = 50, currentPage = 1
        if jsonObj.%IsDefined("pagination") {
            set pagObj = jsonObj.pagination.%Get(0)
            set pageSize = pagObj.pageSize
            set currentPage = pagObj.currentPage
        }
        
        set sql = "SELECT ID, RuleCode, RuleName, RuleType, DrgCode, DeptCode, "
                 _"ThresholdPercent, WarningLevel, Status, CreateTime "
                 _"FROM BS_DRGWarningRule WHERE 1=1"
        
        if ruleType '= "" {
            set sql = sql _ " AND RuleType = '"_ruleType_"'"
        }
        if ruleName '= "" {
            set sql = sql _ " AND RuleName LIKE '%"_ruleName_"%'"
        }
        if status '= "" {
            set sql = sql _ " AND Status = '"_status_"'"
        }
        set sql = sql _ " ORDER BY CreateTime DESC"
        
        set stmt = ##class(%SQL.Statement).%New()
        do stmt.%Prepare(sql)
        set rs = stmt.%Execute()
        
        set count = 0, dataArr = []
        while rs.%Next() {
            set count = count + 1
            continue:(count < ((currentPage-1)*pageSize+1)) || (count > (currentPage*pageSize))
            
            set dataObj = {}
            set dataObj.ruleCode = rs.%Get("RuleCode")
            set dataObj.ruleName = rs.%Get("RuleName")
            set dataObj.ruleType = rs.%Get("RuleType")
            set dataObj.drgCode = rs.%Get("DrgCode")
            set dataObj.deptCode = rs.%Get("DeptCode")
            set dataObj.thresholdPercent = rs.%Get("ThresholdPercent")
            set dataObj.warningLevel = rs.%Get("WarningLevel")
            set dataObj.status = rs.%Get("Status")
            set dataObj.createTime = rs.%Get("CreateTime")
            do dataArr.%Push(dataObj)
        }
        
        set resultObj.result.total = count
        set resultObj.result.rows = dataArr
    }
    catch(e) {
        set resultObj.errorCode = "02010101"
        set resultObj.errorMessage = "查询预警规则失败"
    }
    
    quit resultObj
}
```

### 3.2 保存预警规则（Code: 02010102）

```objectscript
ClassMethod SaveWarningRule(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"保存成功"}
    
    try {
        set params = jsonObj.params.%Get(0)
        set ruleCode = params.ruleCode
        
        // 查询是否存在
        set obj = ##class(User.BSDRGWarningRule).%OpenId(ruleCode)
        if obj = "" {
            set obj = ##class(User.BSDRGWarningRule).%New()
            set obj.RuleCode = ruleCode
        }
        
        set obj.RuleName = params.ruleName
        set obj.RuleType = params.ruleType
        set obj.DrgCode = params.drgCode
        set obj.DeptCode = params.deptCode
        set obj.ThresholdPercent = params.thresholdPercent
        set obj.WarningLevel = params.warningLevel
        set obj.Status = params.status
        set obj.UpdateTime = $ZDateTime($H, 3)
        
        set sc = obj.%Save()
        if $$$ISERR(sc) {
            set resultObj.errorCode = "02010102"
            set resultObj.errorMessage = "保存失败"
        }
    }
    catch(e) {
        set resultObj.errorCode = "02010102"
        set resultObj.errorMessage = "保存预警规则异常"
    }
    
    quit resultObj
}
```

### 3.4 确认/处理预警（Code: 02010104）

```objectscript
ClassMethod ProcessWarning(jsonObj) As %Library.DynamicObject
{
    new (jsonObj)
    set resultObj = {"errorCode":"0","errorMessage":"处理成功"}
    
    try {
        set params = jsonObj.params.%Get(0)
        set warningNo = params.warningNo
        set action = params.action              // confirm/process
        set userId = params.userId
        set processAction = params.processAction
        set remarks = params.remarks
        
        set obj = ##class(User.BSDRGWarningRecord).%OpenId(warningNo)
        if obj = "" {
            set resultObj.errorCode = "02010104"
            set resultObj.errorMessage = "预警记录不存在"
            quit resultObj
        }
        
        if action = "confirm" {
            set obj.WarningStatus = "已确认"
            set obj.ConfirmTime = $ZDateTime($H, 3)
            set obj.ConfirmUser = userId
        }
        elseif action = "process" {
            set obj.WarningStatus = "已处理"
            set obj.ProcessTime = $ZDateTime($H, 3)
            set obj.ProcessUser = userId
            set obj.ProcessAction = processAction
            set obj.Remarks = remarks
        }
        
        set sc = obj.%Save()
        if $$$ISERR(sc) {
            set resultObj.errorCode = "02010104"
            set resultObj.errorMessage = "处理失败"
        }
    }
    catch(e) {
        set resultObj.errorCode = "02010104"
        set resultObj.errorMessage = "处理预警异常"
    }
    
    quit resultObj
}
```

## 4. 前端页面设计

### 4.1 预警中心页面

```typescript
// pages/Warning/Center/index.tsx
import React, { useState } from 'react';
import { Card, Table, Tag, Button, DatePicker, Select, Badge, message } from 'antd';
import { queryWarningRecords, processWarning } from '@/api/warning';
import { useTable } from '@/hooks/useTable';
import type { ColumnsType } from 'antd/es/table';

const { RangePicker } = DatePicker;
const { Option } = Select;

const WarningCenter: React.FC = () => {
  const [selectedRowKeys, setSelectedRowKeys] = useState<React.Key[]>([]);
  
  const { tableProps, refresh } = useTable({
    api: queryWarningRecords,
    defaultParams: { pageSize: 50, currentPage: 1 }
  });

  const columns: ColumnsType<WarningRecord> = [
    {
      title: '预警编号',
      dataIndex: 'warningNo',
      width: 150
    },
    {
      title: '患者信息',
      render: (_, record) => (
        <div>
          <div>{record.psnName}</div>
          <div style={{ fontSize: 12, color: '#999' }}>{record.medcasNo}</div>
        </div>
      )
    },
    {
      title: '科室',
      dataIndex: 'deptName'
    },
    {
      title: '主管医生',
      dataIndex: 'doctorName'
    },
    {
      title: 'DRG',
      dataIndex: 'drgCode',
      render: (text) => <Tag color="blue">{text}</Tag>
    },
    {
      title: '基准费用',
      dataIndex: 'benchmarkCost',
      render: (value) => `¥${value?.toFixed(2)}`
    },
    {
      title: '当前费用',
      dataIndex: 'actualCost',
      render: (value) => `¥${value?.toFixed(2)}`
    },
    {
      title: '偏差率',
      dataIndex: 'deviationRate',
      render: (value) => {
        const color = value > 30 ? 'red' : value > 15 ? 'orange' : 'gold';
        return <Tag color={color}>{value?.toFixed(2)}%</Tag>;
      }
    },
    {
      title: '级别',
      dataIndex: 'warningLevel',
      render: (level) => {
        const colorMap: Record<string, string> = {
          '严重': 'red',
          '警告': 'orange',
          '提示': 'gold'
        };
        return <Badge color={colorMap[level]} text={level} />;
      }
    },
    {
      title: '状态',
      dataIndex: 'warningStatus',
      render: (status) => {
        const colorMap: Record<string, string> = {
          '未确认': 'default',
          '已确认': 'processing',
          '已处理': 'success'
        };
        return <Tag color={colorMap[status]}>{status}</Tag>;
      }
    },
    {
      title: '操作',
      render: (_, record) => (
        <span>
          {record.warningStatus === '未确认' && (
            <Button 
              type="link" 
              size="small"
              onClick={() => handleConfirm(record.warningNo)}
            >
              确认
            </Button>
          )}
          {record.warningStatus === '已确认' && (
            <Button 
              type="link" 
              size="small"
              onClick={() => handleProcess(record.warningNo)}
            >
              处理
            </Button>
          )}
        </span>
      )
    }
  ];

  const handleConfirm = async (warningNo: string) => {
    const result = await processWarning({
      warningNo,
      action: 'confirm',
      userId: 'currentUser'
    });
    if (result.errorCode === '0') {
      message.success('确认成功');
      refresh();
    }
  };

  return (
    <Card title="预警监控中心">
      {/* 统计卡片 */}
      <div style={{ marginBottom: 16, display: 'flex', gap: 16 }}>
        <Card style={{ flex: 1 }}>
          <div style={{ textAlign: 'center' }}>
            <div style={{ fontSize: 24, fontWeight: 'bold', color: '#ff4d4f' }}>12</div>
            <div>今日新增预警</div>
          </div>
        </Card>
        <Card style={{ flex: 1 }}>
          <div style={{ textAlign: 'center' }}>
            <div style={{ fontSize: 24, fontWeight: 'bold', color: '#faad14' }}>8</div>
            <div>待确认预警</div>
          </div>
        </Card>
        <Card style={{ flex: 1 }}>
          <div style={{ textAlign: 'center' }}>
            <div style={{ fontSize: 24, fontWeight: 'bold', color: '#1890ff' }}>5</div>
            <div>待处理预警</div>
          </div>
        </Card>
      </div>

      {/* 筛选条件 */}
      {/* ... */}

      {/* 数据表格 */}
      <Table
        {...tableProps}
        columns={columns}
        rowSelection={{
          selectedRowKeys,
          onChange: setSelectedRowKeys
        }}
        rowKey="warningNo"
      />
    </Card>
  );
};

export default WarningCenter;
```
