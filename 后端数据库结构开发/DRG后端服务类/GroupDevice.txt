/// 设置成功返回结果（封装重复逻辑）
ClassMethod SetSuccessResult(code As %String, message As %String) As %Library.DynamicObject
{
	set resultObj={}
	set resultObj.errorCode=0
	set resultObj.errorMessage=""		

	set result={}
	set result.Code=code
	set result.Desc=Message
	set result.CheckTime=$ZDATETIME($HOROLOG,3)	
    set resultObj.result=result
    
    quit resultObj
}

/// 设置错误返回结果（封装重复逻辑）
ClassMethod SetErrorResult(errorCode As %String, errorMsg As %String) As %Library.DynamicObject
{
	set resultObj={}
	set resultObj.errorCode=errorCode
	set resultObj.errorMessage=errorMsg		

	set result={}
	set result.Code=""
	set result.Message=""
	set result.CheckTime=$ZDATETIME($HOROLOG,3)	
    set resultObj.result=result
    
    quit resultObj
}

/// 校验是否存在 不作为分组规则疾病诊断、手术操作表中
ClassMethod CheckNoRulesList(code As %String) As %String
{
    set iCount = 0
    &sql(SELECT COUNT(*) INTO :iCount 
         FROM HB_DRGBasicDataSub 
         WHERE HBDictionaries_Dr->InsuCode IN ('NoRuleOfDiagnosis', 'NoRuleOfOperation') 
         AND Code = :code)
    
    quit iCount
}

/// Creator:     CM
/// CreatDate:   2026-01-01
/// Description: 分组前校验规则：校验不作为分组规则的疾病诊断和手术操作列表、QY组 检查病例的主要诊断和主要手术/操作是否匹配
///              在分组流程的第一步（确定主要诊断大类MDC）和第二步（确定核心疾病诊断相关组ADRG）中，
///              如果病例的主要诊断或主要手术操作存在于这个列表中，系统就会判定该病例无法进入正常的MDC或ADRG。
///              被过滤掉的病例最终会进入“0000组”（空白病组，即无法入组的病例）
/// code:		 
/// Input:{"params":[{"":""}]}
/// Debug：
ClassMethod CheckRules(jsonObj) As %Library.DynamicObject
{
	try
	{
        // 1. 初始化返回对象（标准化结构）
        set resultObj = ..SetSuccessResult("0","")
        
        // 2. 参数合法性校验（核心前置检查）
        if (jsonObj = "") || (jsonObj.params = "") 
        {
            set resultObj = ..SetErrorResult("-1", "入参异常：缺少params参数或参数为空")
            quit 
        } 
                   
        // 3. 提取入参并设置默认值（避免空值报错）
        set preMsg=""
        set params = jsonObj.params.%Get(0)
        set mainDiagCode = $GET(params.mainDiagnosisCode, "")  // 主诊断代码
        set mainOperationCode = $GET(params.mainOperationCode, "")  // 主手术代码
        
        if (mainDiagCode ="")
        {
		    set resultObj = ..SetErrorResult("-1", "主诊断不能为空！")
		    quit    
	    }	    	    
        
        //1.剔除不作为分组规则的疾病诊断列表、不作为分组规则的手术操作列表	         	         		
	    set iCountA=..CheckNoRulesList(mainDiagCode)
	    set iCountB=..CheckNoRulesList(mainOperationCode)
	    if (MainOperation '="")
    	{
	         if (iCounA>0)
	         {
		         set preMsg= preMsg_" 该主诊断["_MainDiag_MainDiagName_"]存在CHS-DRG不作为分组规则的疾病诊断列表中,无法分组,请检查！"
             }
		 
	         if (iCounB>0)
	         {
		         set preMsg= preMsg_" 该主手术操作["_MainOperation_MainOperationName_"]存在CHS-DRG不作为分组规则的手术操作列表中,无法分组,请检查！"			 
	         }
	         
	         if (iCounA>0)||(iCounB>0)
	         {
		         set resultObj = ..SetErrorResult("-1", preMsg)
		         quit    
		     } 	        
	    }
	    else	
	    {
	         if (iCounA>0)
	         {
		         set preMsg=OutMsg_" 该主诊断["_MainDiag_MainDiagName_"]存在CHS-DRG不作为分组规则的疾病诊断列表中,无法分组！"
		         set resultObj = ..SetErrorResult("-1", preMsg)
		         quit   
             }	    	
	    }	
	}
    catch(e) 
    {
        // 7. 异常处理（增强：包含错误码+详细信息）
        set status = e.AsStatus()
        set errorCode = $piece(##class(%SYSTEM.Status).GetErrorCodes(status), ",")
        set errorMsg = $replace(##class(%SYSTEM.Status).GetErrorText(status,"cn"), "^", " ")
        set errorInfo = "CheckRules分组前校验规则异常：类=" _ $ClassName() _ "，入参=" _ jsonObj.%ToJSON() _ "，错误：" _ errorMsg
        
        // 记录错误日志到数据库
        set errorLogID = ##class(src.Interface.Message).InsErrRecord("04000007", errorInfo, errorMsg)
        set resultObj = ..SetErrorResult("-99", "CheckRules分组前校验规则异常（日志ID："_errorLogID_")："_errorMsg)
    }

    quit resultObj
}

/// Creator:     CM
/// CreatDate:   2026-01-01
/// Description: 第一步：MDC先期分组
/// code:		 
/// Input:{"params":[{"":""}]}
/// Debug：
ClassMethod GroupPreMDC(jsonObj As %Library.DynamicObject) As %Library.DynamicObject
{
	
    try
    {  
        // 1. 初始化返回对象（标准化结构）
        set resultObj = ..SetSuccessResult("0","")
        
        // 2. 参数合法性校验（核心前置检查）
        if (jsonObj = "") || (jsonObj.params = "") 
        {
            set resultObj = ..SetErrorResult("-1", "入参异常：缺少params参数或参数为空")
            quit 
        }              
        // 3. 提取入参并设置默认值（避免空值报错）
        set params = jsonObj.params.%Get(0)
        set mainDiagCode = $GET(params.MainDiagnosisCode, "")  // 主诊断代码
        set sex = $GET(params.Sex, "")                         // 性别
        set age = $GET(params.Age, 0)                          // 年龄（默认0）
        set ageGroupDays = $GET(params.AgeGroupDays, 0)        // 新生儿出生天数（默认0）
        set newbornFlag = $GET(params.NewbornFlag, "0")        // 新生儿标志（默认0）
        set respiratorTime = $GET(params.RespiratorTime, 0)    // 有创呼吸机时长（默认0）
        set ecmoFlag = $GET(params.ECMOFlag, "0")              // ECMO标志（默认0）
        set transplantFlag = $GET(params.TransplantFlag, "0")  // 器官移植标志（默认0）
        set marrowTransplantFlag = $GET(params.MarrowTransplantFlag, "0") // 骨髓移植标志
        set hivFlag = $GET(params.HIVFlag, "0")                // HIV标志（默认0）
        set traumaLevel = $GET(params.TraumaLevel, 0)          // 多发创伤等级（默认0）

        // 4. 日志记录（新增：便于问题排查）
        set checkParams = "主诊断=" _ mainDiagCode _ ",年龄=" _ age _ ",新生儿天数=" _ ageGroupDays
        do ..LogCheckInfo("PreMDC检查开始", checkParams)

        // 5. 先期分组核心逻辑（按优先级排序：高资源消耗优先）
        // 5.1 MDCA - 高资源消耗病例（器官移植/骨髓移植/ECMO/呼吸机≥96h）
        if (transplantFlag = "1") 
        {
            set resultObj = ..SetSuccessResult("MDCA", "入组MDCA：器官移植病例")
        } 
        elseif (marrowTransplantFlag = "1") 
        {
            set resultObj = ..SetSuccessResult("MDCA", "入组MDCA：骨髓移植病例")
        } elseif (ecmoFlag = "1") 
        {
            set resultObj = ..SetSuccessResult("MDCA", "入组MDCA：ECMO治疗病例")
        } elseif (respiratorTime >= 96) 
        {
            set resultObj = ..SetSuccessResult("MDCA", "入组MDCA：有创呼吸机使用时长≥96小时（实际："_respiratorTime_"小时）")
        }
        // 5.2 MDCP - 新生儿病例（出生天数1-28天 或 新生儿标志为1）
        elseif ((ageGroupDays > 0 && ageGroupDays < 29) || (newbornFlag = "1")) 
        {
            set resultObj = ..SetSuccessResult("MDCP", "入组MDCP：新生儿疾病（出生天数："_ageGroupDays_"天）")
        }
        // 5.3 MDCY - HIV感染病例
        elseif (hivFlag = "1") 
        {
            set resultObj = ..SetSuccessResult("MDCY", "入组MDCY：HIV感染相关疾病")
        }
        // 5.4 MDCZ - 多发严重创伤（等级≥2）
        elseif (traumaLevel >= 2) 
        {
            set resultObj = ..SetSuccessResult("MDCZ", "入组MDCZ：多发严重创伤（创伤等级："_traumaLevel_"级）")
        }
        // 5.5 未命中先期分组（返回空编码，进入常规MDC分组）
        else 
        {
		    set (mdcCode,mdcDesc)=""
		    &sql(SELECT HBDictionaries_Dr->InsuCode,HBDictionaries_Dr->InsuDesc into :mdcCode,:mdcDesc 
		         FROM HB_DRGBasicDataSub WHERE HBDictionaries_Dr->Remark='主诊断编码对照表' 
		         AND Code= :mainDiagCode )
		    
		    if (mdcCode'="")
		    {
			    set resultObj = ..SetSuccessResult(mdcCode, "入组"_mdcCode_":"_mdcDesc) 
			}
			else
			{
			    set resultObj = ..SetErrorResult("-1", "通过主诊断 "_mainDiagCode_", MDC分组失败！")	
			}	        	                    
        }

        // 6. 日志记录检查结果
        do ..LogCheckInfo("PreMDC检查结束", "分组结果：" _ resultObj.result.Code _ "，说明：" _ resultObj.result.Message)        
    } 
    catch(e) 
    {
        // 7. 异常处理（增强：包含错误码+详细信息）
        set status = e.AsStatus()
        set errorCode = $piece(##class(%SYSTEM.Status).GetErrorCodes(status), ",")
        set errorMsg = $replace(##class(%SYSTEM.Status).GetErrorText(status,"cn"), "^", " ")
        set errorInfo = "PreMDC分组异常：类=" _ $ClassName() _ "，入参=" _ jsonObj.%ToJSON() _ "，错误：" _ errorMsg
        
        // 记录错误日志到数据库
        set errorLogID = ##class(src.Interface.Message).InsErrRecord("04000007", errorInfo, errorMsg)
        set resultObj = ..SetErrorResult("-99", "MDC先期分组异常（日志ID："_errorLogID_")："_errorMsg)
    }

    quit resultObj
}

/// Creator:     CM
/// CreatDate:   2026-01-01
/// Description: 第二步：ADRG分组 
/// code:		 
/// Input:{"params":[{"":""}]}
/// Debug：
ClassMethod GroupADRG(jsonObj) As %Library.DynamicObject
{
     //ADRG分组顺序
     //ADRG联合分组规则有AC1、AH2、PS1、PS2、PS3、PS4、BB1、CB2、CB3、FB1、FF1、YC1、YR1、YR2、ZZ1     
	 //1.先期分组+联合分组规则--AC1、AH2、PS1、PS2、PS3、PS4、YC1、YR1、YR2、ZZ1
	 //2.先期分组+非联合分组规则--AA1、AA2、AB1、AD1、AE1、AF1、AG1、AG2、AG3、AH1、PB1、PC1、PD1、PJ1、PK1、PR1、PU1、PV1、ZB1、ZC1、ZD1、ZJ1
	 //3.非先期分组+联合分组规则--BB1、CB2、CB3、FB1、FF1...
	 //4.非先期分组+非联合分组规则+主要诊断--...
	 //5.非先期分组+非联合分组规则+主要手术或操作--...
    
    try
    {
        // 1. 初始化返回对象（标准化结构）
        set resultObj = ..SetSuccessResult("0","")
        
        // 2. 参数合法性校验（核心前置检查）
        if (jsonObj = "") || (jsonObj.params = "") 
        {
            set resultObj = ..SetErrorResult("-1", "入参异常：缺少params参数或参数为空")
            quit 
        }	    
        // 3. 提取入参并设置默认值（避免空值报错）
        set params = jsonObj.params.%Get(0)
        set mainDiagCode = $GET(params.MainDiagnosisCode, "")  // 主诊断代码 
        set diseinfo = $GET(params.DiseInfo, "")  // 疾病诊断代码-数组对象 	
        set mainOpCode = $GET(params.MainOperationCode, "")  // 主手术操作代码
        set oprninfo = $GET(params.OprnInfo, "")  // 手术操作代码-数组对象                        
        set respiratorTime = $GET(params.RespiratorTime, 0)    // 有创呼吸机时长（默认0）
        set sex = $GET(params.Sex, "")                         // 性别
        set age = $GET(params.Age, 0)                          // 年龄（默认0）
        set MDC = $GET(params.MDC, 0)                          //MDC主诊断大类分组                
	    
	    //先期分组标志
	    set preMDCFlag=0
	    if ($listfind($lb("MDCA","MDCP","MDCY","MDCZ"), MDC) = 1)
	    {
		    set preMDCFlag=1   
		}       
       
        Set SqlStr = " SELECT Code,Descripts,Identification FROM HB_DRGBasicDataSub WHERE HBDictionaries_Dr->InsuCode='ADRG' "  
                   _ " AND RIGHT(Code, 1)= '"_$EXTRACT(MDC,4)_"'"
	
		set sqlObj=##class(%SQL.Statement).%New(1)
		set sc=sqlObj.%Prepare(sqlStr)
		set list=sqlObj.%Execute()

	    set (adrgCode,adrgDesc)= ""	    
        while list.%Next(sc)
        {
	        set adrgCode= list.%Get("Code")      //ADRG编码
	        set adrgDesc= list.%Get("Descripts") //ADRG名称
	        	        
			//查询当前ADRG分组规则  主要诊断/主要手术或操作
			set (adrg,selectionCriteria,unionFlag)="" //入组规则、联合规则标志			      				         				        
			&sql(SELECT Top 1 ADRG,SelectionCriteria,UnionFlag into :adrg,:selectionCriteria,:unionFlag FROM CB_DRGCoreGroupsList WHERE ADRG=:adrgCode)	        	
	        if (SQLCODE=0)
	        {
		        if (adrg="")
		        {
				    set resultObj = ..SetErrorResult("-1","ADRG核心疾病诊断相关分组表CB_DRGCoreGroupsList未维护:【"_adrgCode_"："_adrgDesc_"】分组规则！")
				    quit	   
			    }
		        
		        if (selectionCriteria="")
		        {
				    set resultObj = ..SetErrorResult("-1","ADRG核心疾病诊断相关分组表CB_DRGCoreGroupsList未维护:【"_adrgCode_"："_adrgDesc_"】入组条件！")
				    quit			        
			    } 
		    }  
		    		         	        	        	               	          
	        //1.先处理--先期分组
	        if (preMDCFlag=1)
	        {
		        //2.处理先期分组--联合分组规则
		        //同时包含以下手术或操作
		        if (adrgCode="AC1") 
		        {			             
		            set (iCount,m)=0		        	
                    // 遍历operationArray数组
                    for j=0:1:oprninfo.%Size()-1 
                    {        
                        // 获取手术操作编码和名称                    
                        set mainFlag = oprninfo.%Get(j).MainFlag
                        set no = oprninfo.%Get(j).No
                        set code = oprninfo.%Get(j).Code
                        set name = oprninfo.%Get(j).Name 
                        
                        //存在不作为分组规则手术及操作代码表中
                        if (..CheckNoRulesList(code)>0)
                        {
	                        continue
	                    }
                                                                                                  
                        &sql(SELECT COUNT(DISTINCT ADRG) into :iCount FROM CB_DRGCoreGroupsList WHERE UnionFlag =1  
                             AND ADRG=:adrgCode AND (MajorProcedure=:code OR SecondaryProcedure=:code OR ThirdlyProcedure=:code))
                        // 如果SQL执行成功且有结果
                        if (SQLCODE = 0) && (iCount > 0) 
                        {
                            set m = m + 1
                        }
                        
                        if (m>1)
                        {
				            set resultObj = ..SetSuccessResult(adrgCode,adrgDesc)
				            quit	                    
	                    }                                                	                    	                                              
                    }			        	        			        			        			       	
			    }
			    //（其他诊断 或 手术或操作1）+ 手术或操作2
			    elseif (adrgCode="AH2") 
			    {              
			       set (m,n)=0			       
			       //手术或操作1 + 手术或操作2
			       for i=0:1:oprninfo.%Size()-1 
			       {
                        // 获取手术或操作编码和名称                    
                        set mainFlag = oprninfo.%Get(i).MainFlag
                        set no = oprninfo.%Get(i).No
                        set code = oprninfo.%Get(i).Code
                        set name = oprninfo.%Get(i).Name
                        
                        //存在不作为分组规则手术及操作代码表中
                        if (..CheckNoRulesList(code)>0)
                        {
	                        continue
	                    }                        	
                        
                        //主要手术或操作标识不为1,序号需要为2表示第二次要手术或操作
                        if (mainFlag '= "1")&&(no="2")
                        {
	                        set secOperationCode = code
	                    }
	                    //主要手术或操作标识不为1,序号需要为3表示第三次要手术或操作
	                    elseif (mainFlag '= "1")&&(no="3")
                        {
	                        set thirdOperationCode = code
	                    }
	                    
	                    //手术或操作1 + 手术或操作2
                        &sql(SELECT COUNT(DISTINCT ADRG) into :m FROM CB_DRGCoreGroupsList WHERE UnionFlag =1 AND ADRG=:adrgCode                                    
                             And SecondaryProcedure=:secOperationCode And ThirdlyProcedure=:thirdOperationCode)
                        // 如果SQL执行成功且有结果
                        if (SQLCODE = 0) && (m > 0) 
                        {
                            set resultObj=..SetSuccessResult(adrgCode,adrgDesc)
                            quit
                        }	                    	                    			       
				    }
			    			    			          				   				    				    				   				    				    
                    //其他诊断 + 手术或操作2		        	
                    for j=0:1:diseinfo.%Size()-1 
                    {
                        // 获取诊断编码和名称                    
                        set mainFlag = diseinfo.%Get(j).MainFlag
                        set no = diseinfo.%Get(j).No
                        set code = diseinfo.%Get(j).Code
                        set name = diseinfo.%Get(j).Name
                        
                        //存在不作为分组规则疾病诊断表中
                        if (..CheckNoRulesList(code)>0)
                        {
	                        continue
	                    }                        
            
                        if (mainFlag '="1") //其他诊断
                        {
                            &sql(SELECT COUNT(DISTINCT ADRG) into :n FROM CB_DRGCoreGroupsList WHERE UnionFlag =1
                                 AND ADRG=:adrgCode and (SecondaryDiagnosis=:code OR ThirdlyDiagnosis=:code) And ThirdlyProcedure=:thirdOperationCode)
                            // 如果SQL执行成功且有结果
                            if (SQLCODE = 0) && (n > 0) 
                            {
				                set resultObj = ..SetSuccessResult(adrgCode,adrgDesc)
				                quit
                            }
	                    }                                                	                    	                                              
                    }			          
				}
				//包含以下主要诊断或其他诊断 
			    elseif ($listfind($lb("PS1","PS2","PS3","PS4","YR1","YR2"), adrgCode) = 1)
			    {
				    //遍历诊断数组
                    for j=0:1:diseinfo.%Size()-1 
                    {
                        // 获取诊断编码和名称                    
                        set mainFlag = diseinfo.%Get(j).MainFlag
                        set no = diseinfo.%Get(j).No
                        set code = diseinfo.%Get(j).Code
                        set name = diseinfo.%Get(j).Name
                        
                        //存在不作为分组规则疾病诊断表中
                        if (..CheckNoRulesList(code)>0)
                        {
	                        continue
	                    } 
	                    	                                          
                        if (mainFlag '="1") //其他诊断
                        {
	                        set n=0
                            &sql(SELECT COUNT(DISTINCT ADRG) into :n FROM CB_DRGCoreGroupsList WHERE UnionFlag =1                                    
                                 AND ADRG=:adrgCode AND (PrincipalDiagnosis=:mainDiagCode OR SecondaryDiagnosis=:code OR ThirdlyDiagnosis=:code))
                            // 如果SQL执行成功且有结果
                            if (SQLCODE = 0) && (n > 0) 
                            {
				                set resultObj = ..SetSuccessResult(adrgCode,adrgDesc)
				                quit
                            }
	                    }                                                	                    	                                              
                    }			
				}
				//包含全部手术或操作
				elseif (adrgCode="YC1")
				{
					//存在手术或操作并且存在诊断符合HIV感染疾病及相关操作,符合YC1-HIV相关疾病的手术
					if (mainOpCode '= "") || (otherDiagArray.%Size()>0)
					{
						//遍历诊断数组
						for j=0:1:diseinfo.%Size()-1
						{
                            // 获取诊断编码和名称                    
                            set mainFlag = diseinfo.%Get(j).MainFlag
                            set no = diseinfo.%Get(j).No
                            set code = diseinfo.%Get(j).Code
                            set name = diseinfo.%Get(j).Name
                        
                            //存在不作为分组规则疾病诊断表中
                            if (..CheckNoRulesList(code)>0)
                            {
	                            continue
	                        } 
	                        
	                        set n=0
                            &sql(SELECT COUNT(DISTINCT ADRG) into :n FROM CB_DRGCoreGroupsList WHERE UnionFlag =1                                     
                                 AND ADRG=:adrgCode AND (PrincipalDiagnosis=:code OR SecondaryDiagnosis=:code OR ThirdlyDiagnosis=:code))
                            // 如果SQL执行成功且有结果
                            if (SQLCODE = 0) && (n > 0) 
                            {
				                set resultObj = ..SetSuccessResult(adrgCode,adrgDesc)
				                quit
                            }	                        	                        	                        
						}																		
					}
				}
				//符合 MDCZ 诊断表,且无手术或操作            
				elseif (adrgCode="ZZ1")
				{
					//主要手术及操作和手术及操作数组对象都为空
					if (mainOpCode="") && (operationArray.%Size()=0)
					{
	                    set n=0
                        &sql(SELECT COUNT(DISTINCT ADRG) into :n FROM CB_DRGCoreGroupsList WHERE UnionFlag =1                                    
                             AND ADRG=:adrgCode AND PrincipalDiagnosis=:mainDiagCode)
                        // 如果SQL执行成功且有结果
                        if (SQLCODE = 0) && (n > 0) 
                        {
				            set resultObj = ..SetSuccessResult(adrgCode,adrgDesc)
				            quit
                        }	
					}
				}
				//包含以下主要诊断
				elseif ($listfind($lb("PR1","PU1","PV1"), adrgCode) = 1)

/// Description: 第四步：DRG分组
/// code:		 
/// Input:{"params":[{"":""}]}
/// Debug：
ClassMethod GroupDRG(jsonObj) As %Library.DynamicObject
{
    ///第4位：表示并发症或其他信息，使用阿拉伯数字（0-9），标记病情复杂程度：
    ///1=表示伴有严重并发症与合并症
    ///3=表示伴有一般并发症与合并症
    ///5=表示不伴有并发症与合并症
    ///7=表示死亡或转院
    ///0=表示小于 17 岁组，儿童病例（<17岁）
    ///9=表示未作区分的情况
    ///QY=歧义（QY）组，指与主要诊断无关的手术病例组，可在多个MDC中出现，检查病例的主要诊断和主要手术/操作是否匹配，如果二者在临床逻辑上不匹配，或者不在同一个主要诊断大类（MDC）里，这个病例就会被判定为“歧义”，送入QY组
    ///0000=指由于疾病诊断编码不规范等原因导致的不能正常入组的病例进入的组。
    ///其他数字表示其他需单独分组的情况。
     	
	try
	{
        // 1. 初始化返回对象（标准化结构）
        set resultObj = ..SetSuccessResult("0","")
        
        // 2. 参数合法性校验（核心前置检查）
        if (jsonObj = "") || (jsonObj.params = "") 
        {
            set resultObj = ..SetErrorResult("-1", "入参异常：缺少params参数或参数为空")
            quit 
        }	    
        // 3. 提取入参并设置默认值（避免空值报错）
        set params = jsonObj.params.%Get(0)
        set adrgCode = $GET(params.ADRG, "")  // ADRG代码 
        set complication = $GET(params.Complication, "")  // 并发症 CC/MCC 	
		set mainDiagCode = $GET(params.MainDiagCode, "")  // 主诊断代码 
        set age = $GET(params.Age, 0)                     // 年龄（默认0）		
		
		set num="9"
		if (age>=17)
		{
		    if (complication="CC")
		    {
		        set num="3"		    	
	    	}
		    elseif (complication="MCC")
	    	{
		        set num="1"		    	
	    	}
	    	elseif (complication="")
		    {
		        set num="5"		    	
		    }	
		}
		else
		{
			set num="0"			
		}

		set drg=adrgCode_num		
		set resultObj=..SetSuccessResult(drg,"")
		
	}
    catch(e) 
    {
        // 7. 异常处理（增强：包含错误码+详细信息）
        set status = e.AsStatus()
        set errorCode = $piece(##class(%SYSTEM.Status).GetErrorCodes(status), ",")
        set errorMsg = $replace(##class(%SYSTEM.Status).GetErrorText(status,"cn"), "^", " ")

/// Creator:     CM

/// Description: 分组器

/// Input:{"params":[{"":""}]}
/// Debug
ClassMethod Device(jsonObj) As %Library.DynamicObject
{
	try
	{
        // 初始化返回对象（标准化结构）
        set resultObj = ..SetSuccessResult("0","")
        
        // 参数合法性校验（核心前置检查）
        if (jsonObj = "") || (jsonObj.params = "") 
        {
            set resultObj = ..SetErrorResult("-1", "入参异常：缺少params参数或参数为空")
            quit 
        }
        	    
        // 提取入参并设置默认值（避免空值报错）
        set params = jsonObj.params.%Get(0)
		set mainDiagCode = $GET(params.MainDiagnosisCode, "")    // 主诊断代码
		set diseinfo = $GET(params.DiseInfo, "")  // 疾病诊断-数组对象 	      		
		set mainOperationCode = $GET(params.MainOperationCode, "")    // 主手术代码
    	set oprninfo = $GET(params.OprnInfo, "")  // 手术操作-数组对象    			
        set sex = $GET(params.Sex, "")                         // 性别
        set age = $GET(params.Age, 0)                          // 年龄（默认0）
        set ageGroupDays = $GET(params.AgeGroupDays, 0)        // 新生儿出生天数（默认0）
        set newbornFlag = $GET(params.NewbornFlag, "0")        // 新生儿标志（默认0）
        set respiratorTime = $GET(params.RespiratorTime, 0)    // 有创呼吸机时长（默认0）
        set ecmoFlag = $GET(params.ECMOFlag, "0")              // ECMO标志（默认0）
        set transplantFlag = $GET(params.TransplantFlag, "0")  // 器官移植标志（默认0）
        set marrowTransplantFlag = $GET(params.MarrowTransplantFlag, "0") // 骨髓移植标志
        set hivFlag = $GET(params.HIVFlag, "0")                // HIV标志（默认0）
        set traumaLevel = $GET(params.TraumaLevel, 0)          // 多发创伤等级（默认0）
		set department = $GET(params.Department, "")           // 科室
		set dischargeType = $GET(params.DischargeType, "")     // 离院方式
					
		set jo={},joRtn={}
		set (MDC,ADRG,DRG,Complication)=""
		         	
		//1.分组流程前 校验入参
        set ja=[]
        set ja.params=[]			
		set jo.MainDiagCode = mainDiagCode	
		set jo.MainOperationCode = mainOperationCode		
        do ja.params.%Push(jo)					
	
		set joRtn=..CheckRules(ja)
		if (joRtn.errorCode '= "0")
		{
            set resultObj = ..SetErrorResult("-1", "进入分组流程前CheckRules校验失败：" + joRtn.errorMessage)
            quit	
		}
		
		//2.第一步：MDC先期分组
		//组织MDC分组入参		
		set jo={}
        set ja=[]
        set ja.params=[]		
        set jo.MainDiagCode = mainDiagnosisCode // 主诊断代码
        set jo.Sex = sex                         // 性别
        set jo.Age = age                         // 年龄（默认0）
        set jo.AgeGroupDays = ageGroupDays        // 新生儿出生天数（默认0）
        set jo.NewbornFlag = newbornFlag        // 新生儿标志（默认0）
        set jo.RespiratorTime = respiratorTime    // 有创呼吸机时长（默认0）
        set jo.ECMOFlag = ecmoFlag              // ECMO标志（默认0）
        set jo.TransplantFlag = transplantFlag  // 器官移植标志（默认0）
        set jo.MarrowTransplantFlag = marrowTransplantFlag // 骨髓移植标志
        set jo.HIVFlag = hivFlag               // HIV标志（默认0）
        set jo.TraumaLevel = traumaLevel          // 多发创伤等级（默认0）               
        do ja.params.%Push(jo)
        
		set joRtn=..GroupPreMDC(ja)
		if (joRtn.errorCode '= "0")
		{
            set resultObj = ..SetErrorResult("-1", "1.MDC先期分组失败：" + joRtn.errorMessage)
            quit	
		}
		else
		{
			set MDC=joRtn.result.Code
		}	
		
		//3.第二步：ADRG分组
		//组织ADRG分组入参
		set jo={}
        set ja=[]
        set ja.params=[]				
        set jo.MainDiagCode = mainDiagCode  // 主诊断代码 
        set jo.DiseInfo = diseinfo   // 疾病诊断-数组对象               
        set jo.MainOperationCode = mainOperationCode   // 主手术操作代码        
        set jo.OprnInfo = oprninfo  // 手术操作代码-数组对象        
        set jo.RespiratorTime = respiratorTime    // 有创呼吸机时长（默认0）
        set jo.Sex = sex                         // 性别
        set jo.Age = age                         // 年龄（默认0）
        set jo.MDC = MDC                         //MDC主诊断大类						
        do ja.params.%Push(jo)				
									
		set joRtn=..GroupADRG(ja)
		if (joRtn.errorCode '= "0")
		{

            quit	
		}
		else
		{
			set ADRG=joRtn.result.Code
		}

		set jo={}
        set ja=[]
        set ja.params=[]			
        set jo.ADRG = ADRG                  // ADRG代码

		//第四步：DRG分组
		set jo={}
        set ja=[]
        set ja.params=[]

        set jo.Complication = complication // 并发症 CC/MCC

        do ja.params.%Push(jo)		
		
		set joRtn=..GroupDRG(ja)

    catch(e)
