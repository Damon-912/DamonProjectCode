/// 接口服务注册配置说明
/// 在CB_MapInterface表中注册以下新增接口服务
/// 
/// 接口Code规范:
/// - 020101xx: 费用预警模块
/// - 020102xx: 盈亏分析模块  
/// - 020103xx: 病案质控模块
/// - 020104xx: 系统管理模块
/// 
/// =====================================================
/// 费用预警模块接口 (020101xx)
/// =====================================================
/// | Code      | 类名               | 方法名           | 描述                |
/// |-----------|--------------------|------------------|---------------------|
/// | 02010101  | User.DRG.Warning   | QueryWarningRules | 查询预警规则列表    |
/// | 02010102  | User.DRG.Warning   | SaveWarningRule  | 保存预警规则        |
/// | 02010103  | User.DRG.Warning   | QueryWarningRecords | 查询预警记录列表 |
/// | 02010104  | User.DRG.Warning   | ProcessWarning   | 处理预警            |
/// | 02010105  | User.DRG.Warning   | DeleteWarningRule| 删除预警规则        |
/// | 02010106  | User.DRG.Warning   | QueryWarningStats| 预警统计分析        |
/// 
/// =====================================================
/// 盈亏分析模块接口 (020102xx)
/// =====================================================
/// | Code      | 类名               | 方法名           | 描述                |
/// |-----------|--------------------|------------------|---------------------|
/// | 02010201  | User.DRG.Profit    | QueryDeptProfit  | 查询科室盈亏        |
/// | 02010202  | User.DRG.Profit    | QueryDoctorProfit| 查询医生盈亏        |
/// | 02010203  | User.DRG.Profit    | QueryDiseaseProfit| 查询病种盈亏       |
/// | 02010204  | User.DRG.Profit    | QueryProfitTrend | 盈亏趋势分析        |
/// | 02010205  | User.DRG.Profit    | RecalculateProfit| 重新计算盈亏        |
/// | 02010206  | User.DRG.Profit    | QueryDeptSummary | 查询科室汇总        |
/// 
/// =====================================================
/// 病案质控模块接口 (020103xx)
/// =====================================================
/// | Code      | 类名               | 方法名           | 描述                |
/// |-----------|--------------------|------------------|---------------------|
/// | 02010301  | User.DRG.QC        | PerformQCCheck   | 执行质控检查        |
/// | 02010302  | User.DRG.QC        | QueryQCIssues    | 查询质控问题列表    |
/// | 02010303  | User.DRG.QC        | ProcessQCIssue   | 处理质控问题        |
/// | 02010304  | User.DRG.QC        | QueryQCStats     | 质控统计报表        |
/// 
/// =====================================================
/// 系统管理模块接口 (020104xx)
/// =====================================================
/// | Code      | 类名               | 方法名           | 描述                |
/// |-----------|--------------------|------------------|---------------------|
/// | 02010401  | User.DRG.System    | QuerySystemUsers | 查询系统用户列表    |
/// | 02010402  | User.DRG.System    | SaveSystemUser   | 保存系统用户        |
/// | 02010403  | User.DRG.System    | QuerySystemMenus | 查询系统菜单        |
/// | 02010404  | User.DRG.System    | QueryInterfaceServices| 查询接口服务    |
/// | 02010405  | User.DRG.System    | SaveInterfaceService| 保存接口服务       |
/// | 02010406  | User.DRG.System    | QuerySystemRoles | 查询系统角色        |
/// | 02010407  | User.DRG.System    | SaveSystemRole   | 保存系统角色        |
/// | 02010408  | User.DRG.System    | GetUserMenus     | 获取用户菜单权限    |
/// | 02010409  | User.DRG.System    | SaveRoleMenu     | 保存角色菜单权限    |
/// | 02010410  | User.DRG.System    | UserLogin        | 用户登录            |
/// | 02010411  | User.DRG.System    | UserLogout       | 用户登出            |
/// 
/// =====================================================
/// 接口调用统一格式
/// =====================================================
/// 
/// 请求格式:
/// {
///     "params": [{
///         "业务参数1": "value1",
///         "业务参数2": "value2",
///         ...
///     }],
///     "code": "02010101",
///     "session": [{
///         "userID": "158",
///         "locID": 1,
///         "groupID": "1",
///         "hospID": "8",
///         "sessionID": "mEezLwofKB",
///         "hospCode": "12344"
///     }]
/// }
/// 
/// 响应格式:
/// {
///     "errorCode": "0",
///     "errorMessage": "",
///     "result": {}
/// }
/// 
/// 错误码说明:
/// - 0: 成功
/// - 1001: 参数错误
/// - 1002: 会话过期
/// - 1003: 权限不足
/// - 2001: 数据不存在
/// - 2002: 数据已存在
/// - 3001: 服务器内部错误
/// 
Class User.DRGInterfaceConfig Extends %RegisteredObject
{

/// 获取接口配置列表

{
    set list = ##class(%ListOfObjects).%New()
    
    // 费用预警模块
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010101", "User.DRG.Warning", "QueryWarningRules", "查询预警规则列表", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010102", "User.DRG.Warning", "SaveWarningRule", "保存预警规则", "A"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010103", "User.DRG.Warning", "QueryWarningRecords", "查询预警记录列表", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010104", "User.DRG.Warning", "ProcessWarning", "处理预警", "U"))
    
    // 盈亏分析模块
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010201", "User.DRG.Profit", "QueryDeptProfit", "查询科室盈亏", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010202", "User.DRG.Profit", "QueryDoctorProfit", "查询医生盈亏", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010203", "User.DRG.Profit", "QueryDiseaseProfit", "查询病种盈亏", "S"))
    
    // 病案质控模块
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010301", "User.DRG.QC", "PerformQCCheck", "执行质控检查", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010302", "User.DRG.QC", "QueryQCIssues", "查询质控问题列表", "S"))
    
    // 系统管理模块
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010401", "User.DRG.System", "QuerySystemUsers", "查询系统用户列表", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010402", "User.DRG.System", "SaveSystemUser", "保存系统用户", "A"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010403", "User.DRG.System", "QuerySystemMenus", "查询系统菜单", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010404", "User.DRG.System", "QueryInterfaceServices", "查询接口服务", "S"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010405", "User.DRG.System", "SaveInterfaceService", "保存接口服务", "A"))
    do list.Insert(##class(User.DRGInterfaceConfig).%New().InitInterface("02010408", "User.DRG.System", "GetUserMenus", "获取用户菜单权限", "S"))
    
    quit list
}

Property Code As %String;
Property ClassName As %String;
Property MethodName As %String;
Property Description As %String;
Property ServiceType As %String;

Method InitInterface(pCode, pClassName, pMethodName, pDescription, pServiceType) As User.DRGInterfaceConfig
{
    set ..Code = pCode
    set ..ClassName = pClassName
    set ..MethodName = pMethodName
    set ..Description = pDescription
    set ..ServiceType = pServiceType
    quit $this
}
}
